-- =========================================
-- Advanced SQL Queries on Single Table
-- Table: students
-- =========================================


-- Step 2: Aggregate Queries

-- 1. Count the total number of students
SELECT COUNT(*) AS total_students
FROM students;

-- Output:
-- total_students = 10


-- 2. Find the average age of students
SELECT AVG(age) AS average_age
FROM students;

-- Output:
-- average_age = 21.4


-- 3. Find the minimum and maximum age
SELECT MIN(age) AS minimum_age,
       MAX(age) AS maximum_age
FROM students;

-- Output:
-- minimum_age = 18
-- maximum_age = 25


-- 4. Count how many students are enrolled in each course
SELECT course, COUNT(*) AS student_count
FROM students
GROUP BY course;

-- Output:
-- course              | student_count
-- -----------------------------------
-- Computer Science   | 4
-- Data Science       | 3
-- Information Tech  | 3


-- 5. Count students whose age is greater than 20
SELECT COUNT(*) AS students_above_20
FROM students
WHERE age > 20;

-- Output:
-- students_above_20 = 6



-- =========================================
-- Step 3: Grouping & Filtering
-- =========================================

-- 1. Display course-wise student count
SELECT course, COUNT(*) AS student_count
FROM students
GROUP BY course;

-- Output:
-- course              | student_count
-- -----------------------------------
-- Computer Science   | 4
-- Data Science       | 3
-- Information Tech  | 3


-- 2. Display only courses where student count is greater than 1
SELECT course, COUNT(*) AS student_count
FROM students
GROUP BY course
HAVING COUNT(*) > 1;

-- Output:
-- course              | student_count
-- -----------------------------------
-- Computer Science   | 4
-- Data Science       | 3
-- Information Tech  | 3


-- 3. Display courses sorted by highest student count first
SELECT course, COUNT(*) AS student_count
FROM students
GROUP BY course
ORDER BY student_count DESC;

-- Output:
-- course              | student_count
-- -----------------------------------
-- Computer Science   | 4
-- Data Science       | 3
-- Information Tech  | 3



-- =========================================
-- Step 4: Ordering & Limiting
-- =========================================

-- 1. Fetch the latest 3 students based on created_at
